import { Card, CardGrid, LinkCard } from '@astrojs/starlight/components';

<CardGrid>
  <Card title="Self-contained modules" icon="puzzle">
    Each module holds its own providers, routes, controllers, models, views, migrations, config and translations in one folder.
  </Card>
  <Card title="Generators" icon="rocket">
    `module:make` creates a module. `module:make-controller`, `module:make-model`, `module:make-migration` and more create the classes inside it.
  </Card>
  <Card title="Enable and disable" icon="setting">
    Disable a module and the package stops registering its providers. Its routes disappear until you enable it again.
  </Card>
  <Card title="Module-aware database commands" icon="seti:db">
    `module:migrate`, `module:seed` and the rollback, reset, refresh and fresh variants run for one module or for all of them.
  </Card>
</CardGrid>

Install the package with Composer:

```bash
composer require nwidart/laravel-modules
```
